# 🐍 Contribution Snake
name: Generate Snake Animation
 
on:
  schedule:
    - cron: "0 0 * * *"   # প্রতিদিন রাত ১২টায় (UTC) অটো-রান হবে
  workflow_dispatch:        # চাইলে ম্যানুয়ালিও রান করা যাবে (Actions ট্যাব থেকে)
  push:
    branches:
      - main                # main ব্রাঞ্চে push হলেও রান হবে
 
jobs:
  generate:
    permissions:
      contents: write
    runs-on: ubuntu-latest
    steps:
      - name: Generate contribution snake SVG
        uses: Platane/snk@v3
        id: snake
        with:
          github_user_name: sabidahmed798
          outputs: |
            dist/github-contribution-grid-snake.svg
            dist/github-contribution-grid-snake-dark.svg?palette=github-dark
 
      - name: Push output to the "output" branch
        uses: crazy-max/ghaction-github-pages@v4
        with:
          target_branch: output
          build_dir: dist
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
